CNM-TESTBED-MIB DEFINITIONS ::= BEGIN

--
-- MIB de pruebas de CNM (designer/testbed). Una tabla que sirve cnm-testbed-agent
-- con pass_persist, en la rama netSnmpPlaypen (.1.3.6.1.4.1.8072.9999.9999), que
-- net-snmp reserva para experimentos. No se usa fuera de un equipo de pruebas.
--

IMPORTS
	MODULE-IDENTITY, OBJECT-TYPE, Integer32, Gauge32, enterprises
		FROM SNMPv2-SMI
	DisplayString
		FROM SNMPv2-TC;

cnmTestbed MODULE-IDENTITY
	LAST-UPDATED "202611210000Z"
	ORGANIZATION "s30labs"
	CONTACT-INFO "cnm-admin"
	DESCRIPTION  "Tabla de pruebas para cnm-testbed."
	::= { enterprises 8072 9999 9999 7 }

cnmTestbedTable OBJECT-TYPE
	SYNTAX      SEQUENCE OF CnmTestbedEntry
	MAX-ACCESS  not-accessible
	STATUS      current
	DESCRIPTION "Filas de prueba. Las sirve cnm-testbed-agent segun su escenario."
	::= { cnmTestbed 1 }

cnmTestbedEntry OBJECT-TYPE
	SYNTAX      CnmTestbedEntry
	MAX-ACCESS  not-accessible
	STATUS      current
	DESCRIPTION "Una fila de prueba."
	INDEX       { cnmTestbedIndex }
	::= { cnmTestbedTable 1 }

CnmTestbedEntry ::= SEQUENCE {
	cnmTestbedIndex  Integer32,
	cnmTestbedLabel  DisplayString,
	cnmTestbedValue  Gauge32
}

cnmTestbedIndex OBJECT-TYPE
	SYNTAX      Integer32 (1..2147483647)
	MAX-ACCESS  not-accessible
	STATUS      current
	DESCRIPTION "Indice de la fila."
	::= { cnmTestbedEntry 1 }

cnmTestbedLabel OBJECT-TYPE
	SYNTAX      DisplayString
	MAX-ACCESS  read-only
	STATUS      current
	DESCRIPTION "Etiqueta de la fila: la descripcion de la instancia en CNM (get_iid)."
	::= { cnmTestbedEntry 2 }

cnmTestbedValue OBJECT-TYPE
	SYNTAX      Gauge32
	MAX-ACCESS  read-only
	STATUS      current
	DESCRIPTION "Valor de la fila."
	::= { cnmTestbedEntry 3 }

END
